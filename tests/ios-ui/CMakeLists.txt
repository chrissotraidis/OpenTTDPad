cmake_minimum_required(VERSION 3.28)

project(OpenTTDPadUITests LANGUAGES OBJC Swift)

set(CMAKE_XCODE_GENERATE_SCHEME ON)

add_executable(OpenTTDPadUITestHost MACOSX_BUNDLE HostMain.m)
target_link_libraries(OpenTTDPadUITestHost PRIVATE "-framework UIKit")
set_target_properties(OpenTTDPadUITestHost PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER org.openttd.OpenTTDPadUITestHost
    MACOSX_BUNDLE_BUNDLE_NAME OpenTTDPadUITestHost
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO
    XCODE_ATTRIBUTE_GENERATE_INFOPLIST_FILE YES
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 17.0
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER org.openttd.OpenTTDPadUITestHost
    XCODE_ATTRIBUTE_SUPPORTED_PLATFORMS iphonesimulator
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY 2
)

add_executable(OpenTTDPadUITests MACOSX_BUNDLE OpenTTDPadUITests.swift)
target_link_libraries(OpenTTDPadUITests PRIVATE "-framework XCTest")
add_dependencies(OpenTTDPadUITests OpenTTDPadUITestHost)
set_target_properties(OpenTTDPadUITests PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER org.openttd.OpenTTDPadUITests
    SUFFIX .xctest
    XCODE_EXPLICIT_FILE_TYPE wrapper.cfbundle
    XCODE_PRODUCT_TYPE com.apple.product-type.bundle.ui-testing
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO
    XCODE_ATTRIBUTE_GENERATE_INFOPLIST_FILE YES
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 17.0
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER org.openttd.OpenTTDPadUITests
    XCODE_ATTRIBUTE_SUPPORTED_PLATFORMS iphonesimulator
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY 2
    XCODE_ATTRIBUTE_TEST_TARGET_NAME OpenTTDPadUITestHost
)
